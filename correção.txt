melhor_elenco(Elenco) :-
    findall(G-C, melhor_goleiro_selecao(G, C), Goleiros),
    findall(Z-C, melhor_zagueiro_selecao(Z, C), Zagueiros),
    findall(LD-C, melhor_lateral_direito_selecao(LD, C), LateraisDireitos),
    findall(LE-C, melhor_lateral_esquerdo_selecao(LE, C), LateraisEsquerdos),
    findall(V-C, melhor_volante_selecao(V, C), Volantes),
    findall(M-C, melhor_meia_selecao(M, C), Meias),
    findall(A-C, melhor_atacante_selecao(A, C), Atacantes),
    append([Goleiros, Zagueiros, LateraisDireitos, LateraisEsquerdos, Volantes, Meias, Atacantes], Elenco).
